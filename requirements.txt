import csv
from fastavro import reader
from google.cloud import storage
import tempfile
import os
 
def download_from_gcs(bucket_name, source_blob_name, destination_file_name):
    client = storage.Client()
    bucket = client.bucket(bucket_name)
    blob = bucket.blob(source_blob_name)
    blob.download_to_filename(destination_file_name)
    print(f"Downloaded {source_blob_name} to {destination_file_name}")
 
def upload_to_gcs(bucket_name, destination_blob_name, source_file_name):
    client = storage.Client()
    bucket = client.bucket(bucket_name)
    blob = bucket.blob(destination_blob_name)
    blob.upload_from_filename(source_file_name)
    print(f"Uploaded {source_file_name} to {destination_blob_name}")
 
def avro_to_csv_gcs(bucket_name, avro_blob_path, csv_blob_path):
    with tempfile.TemporaryDirectory() as temp_dir:
        local_avro = os.path.join(temp_dir, "temp.avro")
        local_csv = os.path.join(temp_dir, "temp.csv")
 
        # Download Avro file from GCS
        download_from_gcs(bucket_name, avro_blob_path, local_avro)
 
        # Convert to CSV
        with open(local_avro, 'rb') as avro_file:
            avro_reader = reader(avro_file)
            records = list(avro_reader)
 
            if not records:
                print("No records in Avro file.")
                return
 
            fieldnames = list(records[0].keys())
 
            with open(local_csv, 'w', newline='') as csv_file:
                csv_writer = csv.DictWriter(csv_file, fieldnames=fieldnames)
                csv_writer.writeheader()
                csv_writer.writerows(records)
 
        # Upload CSV to GCS
        upload_to_gcs(bucket_name, csv_blob_path, local_csv)

from google.cloud import storage

import pandas as pd

def read_avro_from_gcs(bucket_name, avro_blob_path):
    # Create a temporary file to store the Avro data
    with tempfile.NamedTemporaryFile(delete=False, suffix=".avro") as temp_file:
        temp_path = temp_file.name
 
    # Download the Avro file from GCS
    client = storage.Client()
    bucket = client.bucket(bucket_name)
    blob = bucket.blob(avro_blob_path)
    blob.download_to_filename(temp_path)
    print(f"Downloaded {avro_blob_path} to {temp_path}")
 
    # Read Avro into DataFrame
    with open(temp_path, 'rb') as avro_file:
        avro_reader = reader(avro_file)
        records = list(avro_reader)
 
    os.remove(temp_path)  # Clean up
 
    if not records:
        print("No records found.")
        return pd.DataFrame()
 
    return pd.DataFrame(records)

from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime, timedelta
from google.cloud import storage
import fnmatch
import logging
 
# Define default arguments for the DAG
default_args = {
    'owner': 'airflow',
    'retries': 1,
    'retry_delay': timedelta(minutes=5),
}
 
# Define the DAG
with DAG(
    dag_id='gcs_find_oldest_newest_file',
    default_args=default_args,
    start_date=datetime(2025, 7, 1),
    schedule_interval=None,  # Set to your need
    catchup=False,
    tags=['gcs', 'data-engineering'],
) as dag:
 
    def get_oldest_newest_from_gcs(bucket_name: str, wildcard_prefix: str, **kwargs):
        client = storage.Client()
        bucket = client.bucket(bucket_name)
 
        # Extract base prefix (before wildcard) to limit GCS listing scope
        base_prefix = wildcard_prefix.split('*')[0]
 
        # List all blobs under base_prefix
        all_blobs = list(client.list_blobs(bucket, prefix=base_prefix))
        if not all_blobs:
            logging.info("No files found under base prefix.")
            return
 
        # Filter blobs using fnmatch wildcard
        matched_blobs = [blob for blob in all_blobs if fnmatch.fnmatch(blob.name, wildcard_prefix)]
        if not matched_blobs:
            logging.info("No blobs matched the wildcard pattern.")
            return
 
        # Sort by updated timestamp
        matched_blobs.sort(key=lambda b: b.updated)
 
        oldest_file = matched_blobs[0].name
        newest_file = matched_blobs[-1].name
 
        logging.info(f"Oldest file: {oldest_file}")
        logging.info(f"Newest file: {newest_file}")
 
        # Optionally, push to XComs
        kwargs['ti'].xcom_push(key='oldest_file', value=oldest_file)
        kwargs['ti'].xcom_push(key='newest_file', value=newest_file)
 
    get_files_task = PythonOperator(
        task_id='get_oldest_and_newest_files',
        python_callable=get_oldest_newest_from_gcs,
        op_kwargs={
            'bucket_name': 'your-bucket-name',
            'wildcard_prefix': 'your/path/2025*/files/*.parquet'
        },
        provide_context=True,
    )
 
    get_files_task
